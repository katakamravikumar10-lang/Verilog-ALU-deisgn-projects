  reg [15:0] a;
  reg [15:0] b;
  reg [2:0] sel;
  wire [15:0] y;
  
  alu_16bit dut(
    
    .a(a),
    .b(b),
    .sel(sel),
    .y(y)
  );
  
  initial begin
    
    $dumpfile("alu_16bit.vcd");
    $dumpvars(0,tb);

             $time,a,b,sel,y);
  end
  initial begin
    
    a = 16'b1010101001101011;

     #5;

     #5;
    
    sel = 3'b100;
     #5;
    
    sel = 3'b101;
     #5;
    
    sel = 3'b110;
     #5;
    
    sel = 3'b111;
     #5;
    
    $finish;
    
  end
endmodule
