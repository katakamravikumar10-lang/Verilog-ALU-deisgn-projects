module tb;
  
  reg [7:0] a;
  reg [7:0] b;
  reg [2:0] sel;
  wire [7:0] y;
  
  alu_8bit dut(
    
    .a(a),
    .b(b),
    .sel(sel),
    .y(y)
  );
  
  initial begin
    
    $dumpfile("alu_8bit.vcd");
    $dumpvars(0,tb);
    $monitor("time=%0t, a=%b, b=%b, sel=%b, y=%b",
             $time,a,b,sel,y);
  end
  initial begin
    
    a = 8'b10101010;
    b = 8'b00111100;
    
    sel = 3'b000;
    #5;
    
    sel = 3'b001;
     #5;
    
    sel = 3'b010;
     #5;
    
    sel = 3'b011;
     #5;
    
    sel = 3'b100;
     #5;
    
    sel = 3'b101;
     #5;
    
    sel = 3'b110;
     #5;
    
    sel = 3'b111;
     #5;
    
    $finish;
    
  end
endmodule
